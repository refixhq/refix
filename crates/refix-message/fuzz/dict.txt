# libFuzzer dictionary for the FIX framing scanner.
# Tokens the mutator can splice in, so it builds near-valid frames instead of
# discovering the `8=FIX` anchor by chance.

# Delimiter and the frame anchor.
soh="\x01"
begin_string_prefix="8=FIX"

# Complete BeginString fields.
begin_40="8=FIX.4.0\x01"
begin_42="8=FIX.4.2\x01"
begin_44="8=FIX.4.4\x01"
begin_fixt="8=FIXT.1.1\x01"

# Framing tags: the three whose position the frame structure fixes, plus CheckSum.
tag_body_length="9="
tag_msg_type="35="
tag_checksum="10="

# Complete small fields, to help assemble whole frames.
body_length_short="9=5\x01"
msg_type_logon="35=A\x01"
msg_type_heartbeat="35=0\x01"
checksum_field="10=000\x01"

# Session fields commonly following MsgType.
tag_seq_num="34="
tag_sender="49="
tag_target="56="
tag_sending_time="52="
tag_poss_dup="43="
tag_appl_ver_id="1128="

# Length-prefixed data fields, whose values may legally contain SOH.
tag_raw_data_len="95="
tag_raw_data="96="
tag_xml_data_len="212="
tag_xml_data="213="
